# Guard tests: tests that check a rule of this repository rather than one
# module. `vitest --changed` only picks the tests that import a changed file,
# and a guard reads the tree itself (a directory scan, a doc, a generated
# file, a script it runs), so a change that breaks the rule is usually not
# one it imports. scripts/verify.sh therefore runs every file listed here on
# every change, after the related tests.
#
# One path per line, relative to the repository root; `#` starts a comment.
# Only main-lane tests (vitest.config.ts), not *.runtime.test.* files.
#
# What counts is this list on the change's merge base plus any line the
# change adds: a change can add a guard to its own gate, but removing a line
# takes effect only once that removal is on main.

# A room.notice template the desktop has must be published for the phone and
# listed in docs/phone-client-contract.md.
src/shared/__tests__/phoneRoomNotices.test.ts

# No new `slug === '<agent>'` or `case '<agent>':` branches outside the agent
# registry (src/shared/agentIdentity.ts).
src/shared/__tests__/agentSlugBranches.guard.test.ts

# Every English UI string has a Polish one, with the same placeholders.
src/renderer/i18n/__tests__/plLocaleCoverage.test.ts

# docs/api/reference.md is regenerated after an RPC, event or capability change.
scripts/__tests__/genApiReference.os.test.mjs

# Every test file is named for its lane (`*.os.test.*` is integration) and
# runs in it; the guard reads the whole tree, so no change imports it.
scripts/__tests__/testLanes.os.test.mjs

# Every test path a workflow or package.json script names exists.
scripts/__tests__/workflowTestPaths.test.mjs

# Every main IPC handler uses the shared error wrapper; this scans source
# rather than importing new handlers, so changed-test selection misses it.
src/main/ipc/__tests__/wrapHandler.rollout.test.ts

# App-owned session/connector services must survive daemon handler swaps.
src/main/ipc/__tests__/registerHandlers.race.test.ts

# No test loads a JavaScript file that starts with a hashbang: on a CRLF
# checkout Vite misreads it (#2769). It walks the import graph itself.
scripts/__tests__/hashbangImports.test.mjs

# Every vitest config registers the plugin that loads a CRLF hashbang file;
# it reads the config files at the repository root.
scripts/__tests__/hashbangLf.test.mjs

# Every script that launches the app and reads its CDP port from the app log
# sets the test harness opt-in (WMUX_TEST_CDP_PORT=1); it scans scripts/.
scripts/__tests__/testCdpOptIn.guard.test.mjs

# The native hook bridge's packaging contract: every native-hooks.yml step is
# gated on its relevance check, and RELEVANT names what the bridge depends on.
# It reads the workflow, main.rs and hookFlavours.ts as text, so a workflow
# edit is not a change it imports.
scripts/__tests__/buildNativeHooks.os.test.mjs
